import pandas as pd
import mplfinance as mpf
import matplotlib.pyplot as plt

# Sample stock market data
data = {
    'Open':  [110, 108, 106, 104, 102, 100, 98, 96, 94, 92],
    'High':  [112, 109, 107, 105, 103, 101, 99, 97, 95, 94.5],
    'Low':   [107, 105, 103, 101, 99, 97, 95, 93, 91, 86],
    'Close': [108, 106, 104, 102, 100, 98, 96, 94, 92, 94]
}

dates = pd.date_range(start='2026-10-20', periods=10)

df = pd.DataFrame(data, index=dates)

# Hammer pattern detection
def detect_hammer(row):
    body = abs(row['Close'] - row['Open'])
    lower_shadow = min(row['Open'], row['Close']) - row['Low']
    upper_shadow = row['High'] - max(row['Open'], row['Close'])

    return (
        body > 0
        and lower_shadow >= 2 * body
        and upper_shadow <= body * 0.5
    )

# Identify Hammer candles
df['Hammer'] = df.apply(detect_hammer, axis=1)

# Highlight Hammer candle
hammer_points = pd.Series(float('nan'), index=df.index)
hammer_points[df['Hammer']] = df.loc[df['Hammer'], 'Low'] - 1

# Add Hammer markers
ap = mpf.make_addplot(
    hammer_points,
    type='scatter',
    marker='^',
    markersize=150,
    color='blue'
)

# Display candlestick chart
mpf.plot(
    df,
    type='candle',
    style='yahoo',
    title='Hammer Candlestick Pattern Detection',
    ylabel='Stock Price (₹)',
    volume=False,
    addplot=ap,
    figsize=(12, 7),
    tight_layout=True,
    savefig='hammer_chart.png'
)

# Print detection results
print("\nHammer Candlestick Detection Results")
print("------------------------------------")

for date, row in df.iterrows():
    if row['Hammer']:
        print("Hammer Pattern Detected!")
        print("Date:", date.date())
        print("Open:", row['Open'])
        print("High:", row['High'])
        print("Low:", row['Low'])
        print("Close:", row['Close'])

print("\nChart saved as hammer_chart.png")
